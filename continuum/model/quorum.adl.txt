system Quorum — "Runs a research funding call from publication through submission, eligibility screening, blind peer review and panel adjudication against a fixed budget envelope, and reports where every proposal stands."
  actor Applicant [person] | uses: ::Quorum — "Principal investigator. Drafts, submits and withdraws their own proposals; sees only their own."
  actor FundingOperations [team] | owns: ::Quorum, ResearchFunding, ResearchFunding.Adjudication, ResearchFunding.Calls, ResearchFunding.Identity, ResearchFunding.Notification, ResearchFunding.Oversight, ResearchFunding.PeerReview, ResearchFunding.Submission — "The team accountable for operating Quorum."
  actor Operator [person] | uses: ::Quorum — "Platform operator. Runs backups and restores, watches delivery health, retries or skips a failed delivery with a recorded reason, and executes account erasure. Holds no business role."
  actor PanelChair [person] | uses: ::Quorum — "Panel lead. The only participant who records an award or a decline, and only for a scored proposal."
  actor ProgramOfficer [person] | uses: ::Quorum — "Funding body staff. Publishes and closes calls, confirms eligibility, assigns reviewers and chases overdue reviews."
  actor Reviewer [person] | uses: ::Quorum — "Domain expert. Verifies an ORCID identity, declares conflicts, accepts assignments and scores proposals blind."
  deployable QuorumServer — "One process: the HTTP API, the static UI, the event dispatcher, the overdue sweeper and the outbound-message drain."
  domain ResearchFunding (core) — "Competitive research funding: calls, proposals, independent review and panel awards against a budget envelope."

      workflow PanelDecision v1 | deployedAs: ::Quorum.QuorumServer; governedBy: ResearchFunding.HumansDecide; realizes: ResearchFunding.PanelAdjudication; satisfies: ResearchFunding.PanelAdjudication.ChairDecidesScoredOnly, ResearchFunding.PanelAdjudication.EnvelopeNeverExceeded; triggeredBy: ::Quorum.ResearchFunding.Submission.ProposalScored — "Started by ProposalScored.         state { acknowledgement: optional(DeclineAcknowledgement), callId: text, meanScore: decimal(4,2), proposalId: text, requested: ::Quorum.ResearchFunding.Calls.Money, verdict: optional(PanelVerdict), withinEnvelope: optional(boolean) } retention=instance-audit
        trigger = AdjudicationState{ callId: payload.callId, meanScore: payload.meanScore, proposalId: payload.proposalId, requested: payload.requested } unique-by payload.proposalId
        decision committed = match withinEnvelope { none -> false; some w -> w }
        decision isAward = match verdict { none -> false; some v -> v.outcome == PanelOutcome "award" }
        human-task decide
          assign ::Quorum.PanelChair
          escalation "after P14D without a decision the program officer is notified; the task stays open and nothing is decided by timeout"
          cancellation close-on-instance-termination
          payload=DecisionRequest{ callId: callId, meanScore: meanScore, proposalId: proposalId, requested: requested }
          outcome PanelVerdict = result -> verdict=some(result)
          retention=task-audit
        human-task reconsider
          assign ::Quorum.PanelChair
          escalation "the envelope cannot fund this award and never grows (the envelope is immutable and awards are final); the only outcome this task admits is the chair's acknowledged decline; the case stays open, escalating to the officer, until the chair records it"
          cancellation close-on-instance-termination
          payload=DecisionRequest{ callId: callId, meanScore: meanScore, proposalId: proposalId, requested: requested }
          outcome DeclineAcknowledgement = result -> acknowledgement=some(result)
          retention=task-audit
        invoke commit -> AdjudicationService.DecisionApi.CommitAward in=AwardRequest{ amount: requested, callId: callId, proposalId: proposalId } out=result -> withinEnvelope=some(result.committed) policy[retry "bounded"; on-fail=retry]
        invoke decline -> AdjudicationService.DecisionApi.RecordDecline in=DeclineRequest{ callId: callId, proposalId: proposalId } policy[retry "bounded"; on-fail=retry]
        invoke declineRefused -> AdjudicationService.DecisionApi.RecordDecline in=DeclineRequest{ callId: callId, proposalId: proposalId } policy[retry "bounded"; on-fail=retry]
        publish awarded -> ProposalAwarded payload=DecisionData{ actorId: match verdict { none -> ""; some v -> v.actorId }, amount: requested, at: match verdict { none -> instant 1970-01-01T00:00:00Z; some v -> v.at }, callId: callId, outcome: PanelOutcome "award", proposalId: proposalId, rationale: match verdict { none -> ""; some v -> v.rationale } }
        publish declined -> ProposalDeclined payload=DecisionData{ actorId: match verdict { none -> ""; some v -> v.actorId }, amount: decimal(14,2) 0.00, at: match verdict { none -> instant 1970-01-01T00:00:00Z; some v -> v.at }, callId: callId, outcome: PanelOutcome "decline", proposalId: proposalId, rationale: match verdict { none -> ""; some v -> v.rationale } }
        publish declinedRefused -> ProposalDeclined payload=DecisionData{ actorId: match acknowledgement { none -> ""; some a -> a.actorId }, amount: decimal(14,2) 0.00, at: match acknowledgement { none -> instant 1970-01-01T00:00:00Z; some a -> a.at }, callId: callId, outcome: PanelOutcome "decline", proposalId: proposalId, rationale: match acknowledgement { none -> ""; some a -> a.rationale } }
        publish refused -> AwardRefused payload=AwardRefusedData{ actorId: match verdict { none -> ""; some v -> v.actorId }, at: match verdict { none -> instant 1970-01-01T00:00:00Z; some v -> v.at }, callId: callId, proposalId: proposalId, requested: requested }
        T1: decide -> isAward
        T10: refused -> reconsider
        T2: isAward -> commit when result
        T3: isAward -> decline default
        T4: commit -> committed
        T5: committed -> awarded when result
        T6: committed -> refused default
        T7: reconsider -> declineRefused
        T8: decline -> declined
        T9: declineRefused -> declinedRefused
